 119: def _groq_json(messages, max_tokens=800):
 120:     """Call Groq and parse a JSON object, tolerating markdown code fences."""
 121:     text = _groq(messages, max_tokens=max_tokens, temperature=0.3, json_mode=True)
 122:     if not text:
 123:         return None
 124:     text = text.strip()
 125:     if text.startswith("```"):
 126:         text = text.split("\n", 1)[-1]
 127:         if text.rstrip().endswith("```"):
 128:             text = text.rstrip()[:-3]
 129:         text = text.strip()
 130:     try:
 131:         return json.loads(text)
 132:     except json.JSONDecodeError:
 133:         i, j = text.find("{"), text.rfind("}")
 134:         if i != -1 and j > i:
 135:             try:
 136:                 return json.loads(text[i:j + 1])
 137:             except json.JSONDecodeError:
 138:                 return None
 139:         return None

 189: def _serp_shopping(query, num=8):
 190:     """Google Shopping engine — richer product data (image, price, source, link)."""
 191:     if not SERP_KEY:
 192:         raise RuntimeError("SERP_API_KEY is not configured")
 193:     params = {
 194:         "engine": "google_shopping",
 195:         "q": query,
 196:         "api_key": SERP_KEY,
 197:         "num": str(num),
 198:         "gl": "us",
 199:         "hl": "en",
 200:     }
 201:     url = "https://serpapi.com/search?" + urllib.parse.urlencode(params)
 202:     req = urllib.request.Request(url, headers={"User-Agent": UA})
 203:     with urllib.request.urlopen(req, timeout=15) as r:
 204:         data = json.loads(r.read().decode("utf-8"))
 205:     out = []
 206:     for s in data.get("shopping_results", [])[:num]:
 207:         price = s.get("price")
 208:         if not price and s.get("extracted_price"):

 211:             "title": s.get("title", ""),
 212:             "link": s.get("product_link") or s.get("link", "") or "",
 213:             "snippet": s.get("source") or "",
 214:             "price": price or "",

 217:             "rating": s.get("rating") or "",
 218:             "reviews": s.get("reviews") or "",
 219:             "delivery": s.get("delivery") or "",

 227: def _match_products(query):
 228:     """Deterministic catalog search over name, category, tags."""
 229:     q = query.lower()
 230:     prods = _catalog().get("products", [])
 231:     scored = []
 232:     for p in prods:
 233:         hay = " ".join([
 234:             p.get("name", ""),
 235:             p.get("category", ""),
 236:             " ".join(p.get("tags", [])),
 237:         ]).lower()
 238:         score = 0
 239:         if q in hay:
 240:             score += 5
 241:         for word in q.split():
 242:             if word and word in hay:
 243:                 score += 2
 244:         # size/color matches
 245:         for attr in p.get("sizes", []) + p.get("colors", []):
 246:             if attr.lower() in q:
 247:                 score += 2
 248:         if score:
 249:             scored.append((score, p))
 250:     scored.sort(key=lambda x: -x[0])
 251:     return [p for _, p in scored[:5]]
