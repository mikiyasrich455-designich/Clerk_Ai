619|def _price_le(price_str, budget):
620|    """True if a '$xx.xx' price string is at or under budget (best-effort)."""
621|    try:
622|        digits = "".join(ch for ch in str(price_str) if ch.isdigit() or ch == ".")
623|        val = float(digits)
624|        return val <= float(budget)
625|    except (ValueError, TypeError):
626|        return False
